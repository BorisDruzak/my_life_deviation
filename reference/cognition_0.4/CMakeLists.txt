cmake_minimum_required(VERSION 3.20)
project(cognition_contracts VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(COG_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/profile.json" COG_PROFILE)
string(JSON COG_ID GET "${COG_PROFILE}" id)
if(NOT COG_ID MATCHES "^[A-Za-z0-9._-]+$")
  message(FATAL_ERROR "Invalid profile identifier")
endif()
foreach(key fusion_prior_weight fusion_conflict_gap fusion_reliable_quality category_hysteresis discomfort_quadratic discount_per_day)
  string(JSON COG_${key} GET "${COG_PROFILE}" ${key})
  if(NOT COG_${key} MATCHES "^[0-9]+([.][0-9]+)?$")
    message(FATAL_ERROR "Profile coefficient must be a nonnegative decimal")
  endif()
endforeach()
configure_file(profile.hpp.in ${CMAKE_CURRENT_BINARY_DIR}/profile.hpp @ONLY)
find_package(Threads REQUIRED)
enable_testing()
foreach(target contract_tests examples)
  add_executable(${target} ${target}.cpp)
  target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(COG_SANITIZE)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
add_test(NAME cognition_contracts COMMAND contract_tests)
add_test(NAME numerical_examples COMMAND examples)
