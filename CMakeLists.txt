cmake_minimum_required(VERSION 3.20)
project(npc_world VERSION 0.5.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NPC_SANITIZERS "Enable ASan and UBSan" OFF)
option(NPC_COVERAGE "GCC coverage instrumentation" OFF)
option(NPC_STRICT "Treat compiler warnings as errors" ON)

# Boost.JSON is header-compiled through src/json.cpp. Prefer an installed Boost
# 1.83+ so normal developer builds stay fast; if absent, fetch the exact
# upstream 1.83.0 source archive verified by SHA-256. We intentionally do not
# commit hundreds of unmodified upstream headers into this repository.
find_path(NPC_BOOST_INCLUDE_DIR boost/json.hpp)
if(NPC_BOOST_INCLUDE_DIR)
  file(STRINGS "${NPC_BOOST_INCLUDE_DIR}/boost/version.hpp" _boost_version_line
       REGEX "^#define BOOST_VERSION [0-9]+$")
  string(REGEX REPLACE ".*BOOST_VERSION ([0-9]+).*" "\\1" NPC_BOOST_VERSION "${_boost_version_line}")
endif()
if(NOT NPC_BOOST_INCLUDE_DIR OR NPC_BOOST_VERSION LESS 108300)
  include(FetchContent)
  FetchContent_Declare(boost_source
    URL https://archives.boost.io/release/1.83.0/source/boost_1_83_0.tar.bz2
    URL_HASH SHA256=6478edfe2f3305127cffe8caf73ea0176c53769f4bf1585be237eb30798c3b8e
  )
  FetchContent_GetProperties(boost_source)
  if(NOT boost_source_POPULATED)
    FetchContent_Populate(boost_source)
  endif()
  set(NPC_BOOST_INCLUDE_DIR "${boost_source_SOURCE_DIR}")
endif()

add_library(npc_world STATIC src/json.cpp src/math.cpp src/config.cpp src/world.cpp src/validation.cpp src/spatial.cpp src/persistence.cpp src/dynamics.cpp src/physical.cpp src/knowledge.cpp src/social.cpp src/economy.cpp src/action_start.cpp src/action_commit.cpp src/controller.cpp src/policy/types.cpp src/policy/knowledge_query.cpp src/local_view.cpp src/policy/validation.cpp src/policy/commitments.cpp)
target_include_directories(npc_world PUBLIC include)
target_include_directories(npc_world SYSTEM PUBLIC "${NPC_BOOST_INCLUDE_DIR}")
target_compile_definitions(npc_world PUBLIC BOOST_ALL_NO_LIB)
if(MSVC)
 target_compile_options(npc_world PUBLIC /W4 /utf-8 /permissive- /fp:precise)
else()
 target_compile_options(npc_world PUBLIC -Wall -Wextra -Wpedantic -ffp-contract=off)
 if(NPC_STRICT)
  target_compile_options(npc_world PUBLIC -Werror)
 endif()
 if(NPC_SANITIZERS)
  target_compile_options(npc_world PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(npc_world PUBLIC -fsanitize=address,undefined)
 endif()
 if(NPC_COVERAGE)
  target_compile_options(npc_world PUBLIC --coverage -O0 -g)
  target_link_options(npc_world PUBLIC --coverage)
 endif()
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(npc_tests tests/main.cpp tests/math_tests.cpp tests/model_tests.cpp tests/world_tests.cpp tests/social_tests.cpp tests/regression_tests.cpp tests/controller_tests.cpp tests/policy_goals_tests.cpp tests/policy_index_tests.cpp)
 target_link_libraries(npc_tests PRIVATE npc_world)
 target_include_directories(npc_tests PRIVATE tests)
 target_compile_definitions(npc_tests PRIVATE NPC_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
 add_test(NAME core COMMAND npc_tests)
endif()

add_executable(world_sim app/main.cpp)
target_link_libraries(world_sim PRIVATE npc_world)
include(GNUInstallDirs)
install(TARGETS world_sim RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY data/ DESTINATION ${CMAKE_INSTALL_DATADIR}/npc_world)
if(BUILD_TESTING)
 find_package(Python3 COMPONENTS Interpreter QUIET)
 if(Python3_Interpreter_FOUND)
  add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py $<TARGET_FILE:world_sim> ${CMAKE_CURRENT_SOURCE_DIR})
  add_test(NAME autonomy_t0 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/autonomy_cli_test.py $<TARGET_FILE:world_sim> ${CMAKE_CURRENT_SOURCE_DIR})
 endif()
endif()

add_executable(world_api_example app/api_example.cpp)
target_link_libraries(world_api_example PRIVATE npc_world)
